func TestR3ConcreteRepeatedVerifyIgnoresExhaustedBudget(t *testing.T) {
	r, dns, _ := domainFixture(t, nil)
	ctx := context.Background()
	d := addDomain(t, r, "concrete.example.com", "concrete", "one")
	dns.records[d.Name] = d.CnameTarget
	// Model a spent owner budget with ten recent database rows. No
	// traffic burst or external order is generated: exactly two local issuer calls.
	for i := 0; i < 10; i++ {
		if _, err := r.db.Exec(`INSERT INTO domain_issue_attempts VALUES('one','other.example.com',?)`, r.now().Unix()); err != nil {
			t.Fatal(err)
		}
	}
	calls := 0
	r.issuer = issuerFunc(func(_ context.Context, name string, _ PresentChallenge) ([]byte, []byte, error) {
		calls++
		cert, key := fakeCert(t, name, r.now())
		return cert, key, nil
	})
	for i := 0; i < 2; i++ {
		if _, err := r.Verify(ctx, d); err != nil {
			t.Fatal(err)
		}
	}
	var attempts int
	if err := r.db.QueryRow(`SELECT count(*) FROM domain_issue_attempts`).Scan(&attempts); err != nil {
		t.Fatal(err)
	}
	t.Logf("spent owner budget: actual issuer calls=%d durable attempt rows=%d; second call reorders a fresh certificate", calls, attempts)
	if calls != 0 {
		t.Fatal("concrete Verify bypassed exhausted owner budget")
	}
}

func TestR3ReservationLossDuringFailedRenewalReturnsOldKey(t *testing.T) {
	r, _, d := wildcardFixture(t)
	ctx := context.Background()
	if _, err := r.Verify(ctx, d); err != nil {
		t.Fatal(err)
	}
	r.issuer = issuerFunc(func(_ context.Context, name string, _ PresentChallenge) ([]byte, []byte, error) {
		cert, key := fakeCert(t, name, r.now())
		return cert, key, nil
	})
	if _, err := r.Certificate(ctx, &pb.Domain{Name: "dashboard.apps.ehrlich.dev"}); err != nil {
		t.Fatal(err)
	}
	now := r.now().Add(61 * 24 * time.Hour)
	r.now = func() time.Time { return now }
	if _, err := r.Verify(ctx, d); err != nil {
		t.Fatal(err)
	}
	r.issuer = issuerFunc(func(_ context.Context, _ string, _ PresentChallenge) ([]byte, []byte, error) {
		accounts := store.NewAccountStore(r.db)
		if err := accounts.ReleaseSubdomain("one", "dashboard"); err != nil {
			t.Fatal(err)
		}
		if err := accounts.ReserveSubdomain("two", "dashboard"); err != nil {
			t.Fatal(err)
		}
		return nil, nil, fmt.Errorf("bounded fixture renewal failure")
	})
	pair, err := r.Certificate(ctx, &pb.Domain{Name: "dashboard.apps.ehrlich.dev"})
	if err == nil && len(pair.KeyPem) > 0 {
		t.Fatal("failed renewal returned cached key after app reservation changed owner")
	}
}
