package host

import (
	"context"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/ehrlich-b/tunn/internal/domains"
	"github.com/ehrlich-b/tunn/internal/store"
	pb "github.com/ehrlich-b/tunn/pkg/proto/domainv1"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/status"
)

// The public owner API must deliver the fresh proof required by the D1 fix.
func TestR4PublicAPIExposesCompetingProof(t *testing.T) {
	p, _, _ := domainProxyFixture(t)
	db, err := store.InitForTest()
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	const owner = "ad38bf44-0c4a-40ce-bdee-2d8579bf63cc"
	if _, err := db.Exec(`INSERT INTO accounts(id,primary_email,plan,created_at) VALUES(?, 'owner@example.com', 'pro', 1),('attacker','attacker@example.com','pro',1)`, owner); err != nil {
		t.Fatal(err)
	}
	dns := new(hostDomainDNS)
	registry, err := domains.New(db, p.Domain, []byte("01234567890123456789012345678901"), dns, nil)
	if err != nil {
		t.Fatal(err)
	}
	p.domains = registry
	if _, err := registry.Add(context.Background(), &pb.Domain{Name: "*.example.com", Owner: "attacker"}); err != nil {
		t.Fatal(err)
	}
	token := relayJWT(t, p, "owner@example.com")
	handler := p.createHandler()
	call := func(method, path, body string) *httptest.ResponseRecorder {
		request := httptest.NewRequest(method, "https://tunn.test"+path, strings.NewReader(body))
		request.Header.Set("Authorization", "Bearer "+token)
		response := httptest.NewRecorder()
		handler.ServeHTTP(response, request)
		return response
	}
	response := call("POST", "/api/domains", `{"name":"*.apps.example.com","tunnel":"dashboard"}`)
	if response.Code != 409 {
		t.Fatalf("expected pending conflict, got %d %s", response.Code, response.Body)
	}
	var target string
	if err := db.QueryRow(`SELECT target FROM domain_reclaim_requests WHERE owner=? AND name='*.apps.example.com'`, owner).Scan(&target); err != nil {
		t.Fatal(err)
	}
	_, internalError := registry.Add(context.Background(), &pb.Domain{Name: "*.apps.example.com", Owner: owner, Tunnel: "dashboard"})
	if status.Code(internalError) != codes.AlreadyExists || !strings.Contains(internalError.Error(), target) {
		t.Fatalf("backend omitted proof: %v", internalError)
	}
	listing := call("GET", "/api/domains", "")
	t.Logf("owner POST: status=%d body=%q; owner GET: status=%d body=%q; backend=%v", response.Code, response.Body.String(), listing.Code, listing.Body.String(), internalError)
	if !strings.Contains(response.Body.String(), target) || !strings.Contains(response.Body.String(), "_tunn-verification.apps.example.com") {
		t.Fatalf("public API discarded the unique competing DNS proof; CLI cannot perform reclaim (target also absent from list=%v)", !strings.Contains(listing.Body.String(), target))
	}
	// Publishing the delivered target allows the same public Add to reclaim.
	dns.target = target
	reclaimed := call("POST", "/api/domains", `{"name":"*.apps.example.com","tunnel":"dashboard"}`)
	if reclaimed.Code != 201 {
		t.Fatalf("proved reclaim failed: %d %s", reclaimed.Code, reclaimed.Body)
	}
}
